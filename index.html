import pygame import random import sys # ========================================== # 1. 遊戲初始化與常數設定 # ========================================== pygame.init() WIDTH, HEIGHT = 800, 600 GRID_SIZE = 20 COLS, ROWS = WIDTH // GRID_SIZE, HEIGHT // GRID_SIZE # 顏色定義 BLACK = (20, 20, 20) GRID_COLOR = (40, 40, 40) GREEN = (0, 200, 0) LIGHT_GREEN = (50, 255, 50) RED = (200, 0, 0) LIGHT_RED = (255, 50, 50) YELLOW = (255, 215, 0) CYAN = (0, 255, 255) WHITE = (255, 255, 255) GRAY = (100, 100, 100) HOVER_GRAY = (150, 150, 150) BLUE = (50, 100, 255) # 護盾道具顏色 SHIELD_AURA = (100, 200, 255) # 護盾無敵發光顏色 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("貪吃蛇大逃殺：極速迴廊 - 第十組") clock = pygame.time.Clock() font_small = pygame.font.Font(None, 36) font_large = pygame.font.Font(None, 64) # 遊戲狀態定義 STATE_MENU = 0 STATE_PLAYING = 1 STATE_GAMEOVER = 2 STATE_WIN = 3 STATE_PAUSED = 4 # ========================================== # 2. 貪婪追擊 AI：直線追擊 + 穿牆距離判斷 # ========================================== def get_greedy_next_step(start, target): x, y = start tx, ty = target dx, dy = 0, 0 # 計算 X 軸最短距離（支援判斷穿牆是否比較近） dist_x = tx - x if abs(dist_x) > COLS / 2: dist_x = dist_x - COLS if dist_x > 0 else dist_x + COLS # 計算 Y 軸最短距離（支援判斷穿牆是否比較近） dist_y = ty - y if abs(dist_y) > ROWS / 2: dist_y = dist_y - ROWS if dist_y > 0 else dist_y + ROWS # 優先往距離較遠的那一軸移動，拉近距離 if abs(dist_x) > abs(dist_y): dx = 1 if dist_x > 0 else -1 elif abs(dist_y) > 0: dy = 1 if dist_y > 0 else -1 else: dx = 1 if dist_x > 0 else (-1 if dist_x < 0 else 0) return [(x + dx) % COLS, (y + dy) % ROWS] # ========================================== # 3. 繪圖與物件生成工具 # ========================================== def draw_grid(): for x in range(0, WIDTH, GRID_SIZE): pygame.draw.line(screen, GRID_COLOR, (x, 0), (x, HEIGHT)) for y in range(0, HEIGHT, GRID_SIZE): pygame.draw.line(screen, GRID_COLOR, (0, y), (WIDTH, y)) def draw_button(text, x, y, w, h, mouse_pos): rect = pygame.Rect(x, y, w, h) is_hovered = rect.collidepoint(mouse_pos) color = HOVER_GRAY if is_hovered else GRAY pygame.draw.rect(screen, color, rect, border_radius=10) pygame.draw.rect(screen, WHITE, rect, 2, border_radius=10) text_surf = font_small.render(text, True, WHITE) text_rect = text_surf.get_rect(center=rect.center) screen.blit(text_surf, text_rect) return rect, is_hovered def spawn_item(avoid_list1, avoid_list2=None): while True: item = [random.randint(1, COLS - 2), random.randint(1, ROWS - 2)] overlap = (item in avoid_list1) or (avoid_list2 and item in avoid_list2) if not overlap: return item def init_game_vars(): player_snake = [[10, 10], [9, 10], [8, 10]] player_dir = (1, 0) enemy_snakes = [ [[COLS - 5, 5], [COLS - 4, 5], [COLS - 3, 5], [COLS - 2, 5]], [[5, ROWS - 5], [4, ROWS - 5], [3, ROWS - 5], [2, ROWS - 5]] ] food = spawn_item(player_snake, enemy_snakes[0]) shield_item = spawn_item(player_snake, [food]) score = 0 exit_pos = None enemy_move_counter = 0 has_shield = False invincible_timer = 0 dash_speed_frames = 0 return (player_snake, player_dir, enemy_snakes, food, shield_item, score, exit_pos, enemy_move_counter, has_shield, invincible_timer, dash_speed_frames) # ========================================== # 4. 主遊戲迴圈 # ========================================== def main(): state = STATE_MENU last_move_time = 0 target_score = 5 (player_snake, player_dir, enemy_snakes, food, shield_item, score, exit_pos, enemy_move_counter, has_shield, invincible_timer, dash_speed_frames) = init_game_vars() next_dir = player_dir while True: mouse_pos = pygame.mouse.get_pos() mouse_clicked = False # --- 事件處理 --- for event in pygame.event.get(): if event.type == pygame.QUIT: pygame.quit() sys.exit() if event.type == pygame.MOUSEBUTTONDOWN and event.button == 1: mouse_clicked = True if event.type == pygame.KEYDOWN: # ESC 暫停切換 if event.key == pygame.K_ESCAPE: if state == STATE_PLAYING: state = STATE_PAUSED elif state == STATE_PAUSED: state = STATE_PLAYING if state == STATE_PLAYING: # 方向控制（使用 next_dir 防止同幀快速雙按導致回頭自殺） if (event.key == pygame.K_UP or event.key == pygame.K_w) and player_dir != (0, 1): next_dir = (0, -1) elif (event.key == pygame.K_DOWN or event.key == pygame.K_s) and player_dir != (0, -1): next_dir = (0, 1) elif (event.key == pygame.K_LEFT or event.key == pygame.K_a) and player_dir != (1, 0): next_dir = (-1, 0) elif (event.key == pygame.K_RIGHT or event.key == pygame.K_d) and player_dir != (-1, 0): next_dir = (1, 0) # 空白鍵脫皮衝刺 if event.key == pygame.K_SPACE and len(player_snake) > 3: player_snake.pop() if score > 0: score -= 1 dash_speed_frames = 15 # --- 畫面清空與背景網格 --- screen.fill(BLACK) draw_grid() # --- 狀態 1：主畫面 --- if state == STATE_MENU: title_text = font_large.render("SNAKE ESCAPE", True, LIGHT_GREEN) screen.blit(title_text, (WIDTH // 2 - title_text.get_width() // 2, HEIGHT // 2 - 100)) subtitle = font_small.render("Group 10 Project", True, WHITE) screen.blit(subtitle, (WIDTH // 2 - subtitle.get_width() // 2, HEIGHT // 2 - 40)) _, play_hovered = draw_button("PLAY", WIDTH // 2 - 100, HEIGHT // 2 + 20, 200, 50, mouse_pos) if mouse_clicked and play_hovered: (player_snake, player_dir, enemy_snakes, food, shield_item, score, exit_pos, enemy_move_counter, has_shield, invincible_timer, dash_speed_frames) = init_game_vars() next_dir = player_dir state = STATE_PLAYING _, quit_hovered = draw_button("QUIT", WIDTH // 2 - 100, HEIGHT // 2 + 85, 200, 50, mouse_pos) if mouse_clicked and quit_hovered: pygame.quit() sys.exit() # --- 狀態 2 & 3：遊玩中與暫停狀態 --- elif state == STATE_PLAYING or state == STATE_PAUSED: if state == STATE_PLAYING: if invincible_timer > 0: invincible_timer -= 1 # 計算移動延遲（衝刺時為 30ms 極速，平時隨分數從 180ms 遞減至最快 40ms） current_delay = 30 if dash_speed_frames > 0 else max(40, 180 - (score * 20)) current_time = pygame.time.get_ticks() if current_time - last_move_time > current_delay: if dash_speed_frames > 0: dash_speed_frames -= 1 player_dir = next_dir # 玩家移動與穿牆 new_x = (player_snake[0][0] + player_dir[0]) % COLS new_y = (player_snake[0][1] + player_dir[1]) % ROWS new_head = [new_x, new_y] player_snake.insert(0, new_head) # 吃鑰匙 if new_head == food: score += 1 if score < target_score: food = spawn_item(player_snake, [shield_item]) else: food = [-1, -1] exit_pos = [COLS // 2, 0] else: player_snake.pop() # 吃護盾 if new_head == shield_item: has_shield = True shield_item = [-1, -1] # 抵達出口勝利 if exit_pos and new_head == exit_pos: state = STATE_WIN # 敵方貪婪 AI 移動 enemy_move_counter += 1 if enemy_move_counter % 3 != 0: for e_snake in enemy_snakes: next_step = get_greedy_next_step(e_snake[0], player_snake[0]) e_snake.insert(0, next_step) e_snake.pop() # 碰撞判定 collision = False if new_head in player_snake[1:]: collision = True for e_snake in enemy_snakes: if new_head in e_snake or any(seg in e_snake for seg in player_snake): collision = True if collision: if invincible_timer > 0: pass # 無敵狀態免疫 elif has_shield: has_shield = False invincible_timer = 90 # 1.5 秒無敵時間 else: state = STATE_GAMEOVER last_move_time = current_time # 繪製遊戲物件 if score < target_score: pygame.draw.rect(screen, YELLOW, (food[0] * GRID_SIZE, food[1] * GRID_SIZE, GRID_SIZE, GRID_SIZE)) elif exit_pos: pygame.draw.rect(screen, CYAN, (exit_pos[0] * GRID_SIZE, exit_pos[1] * GRID_SIZE, GRID_SIZE, GRID_SIZE)) if shield_item != [-1, -1]: pygame.draw.rect(screen, BLUE, (shield_item[0] * GRID_SIZE, shield_item[1] * GRID_SIZE, GRID_SIZE, GRID_SIZE), border_radius=5) for e_snake in enemy_snakes: for i, pos in enumerate(e_snake): color = LIGHT_RED if i == 0 else RED pygame.draw.rect(screen, color, (pos[0] * GRID_SIZE, pos[1] * GRID_SIZE, GRID_SIZE - 1, GRID_SIZE - 1)) for i, pos in enumerate(player_snake): if invincible_timer > 0 and (invincible_timer // 5) % 2 == 0: color = SHIELD_AURA else: color = LIGHT_GREEN if i == 0 else GREEN pygame.draw.rect(screen, color, (pos[0] * GRID_SIZE, pos[1] * GRID_SIZE, GRID_SIZE - 1, GRID_SIZE - 1)) # 左上角 UI 資訊 shield_text = " [SHIELD READY]" if has_shield else "" status_text = font_small.render(f"Keys: {score}/{target_score}{shield_text}", True, WHITE) screen.blit(status_text, (10, 10)) # --- ESC 暫停選單覆蓋層 --- if state == STATE_PAUSED: overlay = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA) overlay.fill((0, 0, 0, 180)) screen.blit(overlay, (0, 0)) pause_text = font_large.render("PAUSED", True, WHITE) screen.blit(pause_text, (WIDTH // 2 - pause_text.get_width() // 2, HEIGHT // 2 - 80)) _, cont_hovered = draw_button("CONTINUE", WIDTH // 2 - 100, HEIGHT // 2 + 10, 200, 50, mouse_pos) if mouse_clicked and cont_hovered: state = STATE_PLAYING _, p_quit_hovered = draw_button("QUIT", WIDTH // 2 - 100, HEIGHT // 2 + 75, 200, 50, mouse_pos) if mouse_clicked and p_quit_hovered: state = STATE_MENU # --- 狀態 4：結算畫面 --- elif state == STATE_GAMEOVER or state == STATE_WIN: end_text = font_large.render( "GAME OVER" if state == STATE_GAMEOVER else "SUCCESS!", True, LIGHT_RED if state == STATE_GAMEOVER else CYAN ) screen.blit(end_text, (WIDTH // 2 - end_text.get_width() // 2, HEIGHT // 2 - 100)) score_text = font_small.render(f"Final Keys Collected: {score}/{target_score}", True, WHITE) screen.blit(score_text, (WIDTH // 2 - score_text.get_width() // 2, HEIGHT // 2 - 40)) _, new_hovered = draw_button("NEW GAME", WIDTH // 2 - 100, HEIGHT // 2 + 20, 200, 50, mouse_pos) if mouse_clicked and new_hovered: (player_snake, player_dir, enemy_snakes, food, shield_item, score, exit_pos, enemy_move_counter, has_shield, invincible_timer, dash_speed_frames) = init_game_vars() next_dir = player_dir state = STATE_PLAYING _, end_quit_hovered = draw_button("QUIT", WIDTH // 2 - 100, HEIGHT // 2 + 85, 200, 50, mouse_pos) if mouse_clicked and end_quit_hovered: state = STATE_MENU pygame.display.flip() clock.tick(60) if __name__ == "__main__": main()
